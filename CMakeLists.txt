cmake_minimum_required(VERSION 3.13)
project(netguard C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(NG_SIM_ONLY    "Build without an OS packet backend (simulator + GUI only)" OFF)
option(NG_NO_RESOURCES "Windows: build without the icon/manifest resources (fallback if windres fails)" OFF)

set(NG_SRC src/main.c src/engine.c src/util.c src/delayq.c src/web.c src/sim.c)
if(NG_SIM_ONLY)
  list(APPEND NG_SRC src/backend_stub.c)
elseif(WIN32)
  list(APPEND NG_SRC src/backend_win.c)
else()
  list(APPEND NG_SRC src/backend_linux.c)
endif()

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-O2 -Wall -Wextra)
endif()

function(ng_setup tgt)
  if(WIN32)
    target_compile_definitions(${tgt} PRIVATE _WIN32_WINNT=0x0601 WIN32_LEAN_AND_MEAN __USE_MINGW_ANSI_STDIO=1)
    target_link_libraries(${tgt} ws2_32 winmm iphlpapi shell32 user32 advapi32)
    if(NOT NG_SIM_ONLY)
      if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(WD_ARCH x64)
      else()
        set(WD_ARCH x86)
      endif()
      target_include_directories(${tgt} PRIVATE ${WINDIVERT_DIR}/include)
      target_link_directories(${tgt} PRIVATE ${WINDIVERT_DIR}/${WD_ARCH})
      target_link_libraries(${tgt} WinDivert)
    endif()
  else()
    find_package(Threads REQUIRED)
    target_link_libraries(${tgt} Threads::Threads)
    if(NOT NG_SIM_ONLY)
      find_package(PkgConfig REQUIRED)
      pkg_check_modules(NFQ REQUIRED libnetfilter_queue)
      target_include_directories(${tgt} PRIVATE ${NFQ_INCLUDE_DIRS})
      target_link_libraries(${tgt} ${NFQ_LIBRARIES})
    endif()
  endif()
endfunction()

if(WIN32)
  # -DWINDIVERT_DIR=C:/WinDivert-2.2.2-A  (contains include/ and x64/ or x86/)
  set(WINDIVERT_DIR "" CACHE PATH "Extracted WinDivert package")
  set(NG_WIN_SRC src/winutil.c src/winsvc.c src/tray_win.c)

  # netguard.exe  : GUI program (no console window): tray icon, Windows service, installer shortcuts
  add_executable(netguard WIN32 ${NG_SRC} ${NG_WIN_SRC})
  # netguardc.exe : console program for running in cmd (logs on stdout, Ctrl+C stops it)
  add_executable(netguardc ${NG_SRC} ${NG_WIN_SRC})
  target_compile_definitions(netguardc PRIVATE NG_CONSOLE_BUILD=1)
  ng_setup(netguard)
  ng_setup(netguardc)

  if(NOT NG_NO_RESOURCES)
    enable_language(RC)
    set(NG_RES_DIR ${CMAKE_SOURCE_DIR}/res)
    configure_file(res/netguard.rc.in ${CMAKE_BINARY_DIR}/netguard.rc @ONLY)
    target_sources(netguard  PRIVATE ${CMAKE_BINARY_DIR}/netguard.rc)
    target_sources(netguardc PRIVATE ${CMAKE_BINARY_DIR}/netguard.rc)
  endif()
else()
  add_executable(netguard ${NG_SRC})
  ng_setup(netguard)
endif()
